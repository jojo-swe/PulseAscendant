cmake_minimum_required(VERSION 3.24)
project(PulseAscendant VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(REMIX_BUILD_APP "Build SDL/OpenGL application" ON)
option(REMIX_RUNTIME_TESTS "Enable integration tests requiring a working display" OFF)
option(REMIX_BUILD_RT_PROBE "Build optional Vulkan capability probe for Act II development" OFF)
if(REMIX_BUILD_RT_PROBE)
 find_package(Vulkan 1.2 REQUIRED)
 add_executable(remix_rt_probe tools/rt_probe.cpp)
 target_link_libraries(remix_rt_probe PRIVATE Vulkan::Vulkan)
endif()
include(CTest)
add_library(remix_core src/timeline.cpp src/audio_transport.cpp src/scene_math.cpp src/options.cpp)
target_include_directories(remix_core PUBLIC src)
target_compile_definitions(remix_core PRIVATE REMIX_VERSION="${PROJECT_VERSION}")
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE remix_core)
target_compile_definitions(core_tests PRIVATE REMIX_MUSIC="${CMAKE_CURRENT_SOURCE_DIR}/assets/music/luminous-flow-act1.wav")
add_test(NAME core COMMAND core_tests)
add_executable(act3_plan_test tests/act3_plan_test.cpp)
target_link_libraries(act3_plan_test PRIVATE remix_core)
add_test(NAME act3_plan COMMAND act3_plan_test)
if(REMIX_BUILD_APP)
 include(FetchContent)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TESTS OFF CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/829a65d769d935c4852f8159e964312c0957260a URL_HASH SHA256=48e15cf1a6928453d77afa955d1fdfbbca9169e2bc3d76d111acb9b8bb076d66 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(SDL3)
 # SDL documents this override for keeping workstation paths out of assertion strings.
 target_compile_definitions(SDL3-static PRIVATE "$<$<CONFIG:Release>:SDL_ASSERT_FILE=\"SDL3\">")
 add_library(remix_audio src/audio.cpp)
 target_link_libraries(remix_audio PUBLIC remix_core SDL3::SDL3-static)
 add_executable(audio_tests tests/audio_tests.cpp)
 target_link_libraries(audio_tests PRIVATE remix_audio)
 target_compile_definitions(audio_tests PRIVATE REMIX_MUSIC="${CMAKE_CURRENT_SOURCE_DIR}/assets/music/second-reality-three-acts.wav")
 add_test(NAME audio COMMAND audio_tests)
 set_tests_properties(audio PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
 find_package(OpenGL REQUIRED)
 find_path(REMIX_VULKAN_INCLUDE_DIR vulkan/vulkan.h HINTS "$ENV{VULKAN_SDK}/Include")
 if(NOT REMIX_VULKAN_INCLUDE_DIR)
  FetchContent_Declare(vulkan_headers URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/tar.gz/refs/tags/v1.4.341 URL_HASH SHA256=8876aad926b2e72bdefddc34885472d5332e2d20aa3ed57313466f0bc982cf3f DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(vulkan_headers)
  set(REMIX_VULKAN_INCLUDE_DIR "${vulkan_headers_SOURCE_DIR}/include")
 endif()
 add_executable(PulseAscendant src/main.cpp src/app.cpp src/renderer.cpp src/hardware_rt.cpp)
 if(WIN32)
  enable_language(RC)
  target_sources(PulseAscendant PRIVATE src/pulse-ascendant.rc)
  target_include_directories(PulseAscendant PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding")
 endif()
 target_include_directories(PulseAscendant PRIVATE "${REMIX_VULKAN_INCLUDE_DIR}")
 target_link_libraries(PulseAscendant PRIVATE remix_audio OpenGL::GL)
 add_test(NAME show_info COMMAND PulseAscendant --show-info)
 set_tests_properties(show_info PROPERTIES PASS_REGULAR_EXPRESSION "\"duration\":300")
 if(REMIX_RUNTIME_TESTS)
  # Software OpenGL drivers can spend longer compiling the scene shaders than
  # running the lifecycle itself. Keep its 15 s event limit, but allow startup.
  set(REMIX_DISPLAY_TEST_TIMEOUT 180 CACHE STRING "Display test timeout including shader compilation (seconds)")
  add_test(NAME incomplete_control COMMAND ${CMAKE_COMMAND} "-DAPP=$<TARGET_FILE:PulseAscendant>" "-DSTARTUP_TIMEOUT=${REMIX_DISPLAY_TEST_TIMEOUT}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/short_control_test.cmake")
  add_test(NAME lifecycle COMMAND PulseAscendant --self-test --duration 15)
  math(EXPR display_test_outer_timeout "${REMIX_DISPLAY_TEST_TIMEOUT} + 5")
  set_tests_properties(incomplete_control lifecycle PROPERTIES TIMEOUT ${display_test_outer_timeout})
 endif()
 # Shaders are runtime sources. Synchronize them even on a shader-only build
 # where the executable does not need to be relinked.
 add_custom_target(remix_assets COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:PulseAscendant>/assets")
 add_dependencies(PulseAscendant remix_assets)
 install(TARGETS PulseAscendant RUNTIME DESTINATION .)
 install(DIRECTORY assets DESTINATION . PATTERN "luminous-flow.wav" EXCLUDE PATTERN "luminous-flow-act1.wav" EXCLUDE PATTERN "adventurous-uplift.wav" EXCLUDE PATTERN "rigid-arpeggios.wav" EXCLUDE PATTERN "second-reality-two-acts.wav" EXCLUDE)
 install(DIRECTORY licenses DESTINATION . OPTIONAL)
 install(FILES README.md DESTINATION . OPTIONAL)
 install(FILES LICENSE DESTINATION .)
 install(FILES music/asset-receipt.json music/analysis.json music/asset-receipt-act2.json music/analysis-act2.json music/asset-receipt-act3.json music/analysis-act3.json music/show-audio.json music/direction.json DESTINATION music)
 install(FILES docs/README.sv.md docs/SOURCE_MAP.md docs/VERIFICATION.md docs/VERIFICATION-v0.2.0.md docs/VERIFICATION-v0.3.0.md docs/QUALITY_REVIEW.md docs/QUALITY_REVIEW-v0.3.0.md docs/ACT_II.md docs/ACT_III.md docs/VARIATION_REWORK.md docs/ACT_I_VARIATION.md docs/ACT_II_VARIATION.md docs/ACT_III_VARIATION.md DESTINATION docs OPTIONAL)
 install(DIRECTORY docs/images docs/evidence DESTINATION docs OPTIONAL)
endif()
