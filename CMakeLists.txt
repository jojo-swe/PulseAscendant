cmake_minimum_required(VERSION 3.24)
project(SecondRealityRemix VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(REMIX_BUILD_APP "Build SDL/OpenGL application" ON)
option(REMIX_RUNTIME_TESTS "Enable integration tests requiring a working display" OFF)
include(CTest)
add_library(remix_core src/timeline.cpp src/audio_transport.cpp src/scene_math.cpp src/options.cpp)
target_include_directories(remix_core PUBLIC src)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE remix_core)
target_compile_definitions(core_tests PRIVATE REMIX_MUSIC="${CMAKE_CURRENT_SOURCE_DIR}/assets/music/luminous-flow.wav")
add_test(NAME core COMMAND core_tests)
if(REMIX_BUILD_APP)
 include(FetchContent)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TESTS OFF CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/829a65d769d935c4852f8159e964312c0957260a URL_HASH SHA256=48e15cf1a6928453d77afa955d1fdfbbca9169e2bc3d76d111acb9b8bb076d66 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(SDL3)
 add_library(remix_audio src/audio.cpp)
 target_link_libraries(remix_audio PUBLIC remix_core SDL3::SDL3-static)
 add_executable(audio_tests tests/audio_tests.cpp)
 target_link_libraries(audio_tests PRIVATE remix_audio)
 target_compile_definitions(audio_tests PRIVATE REMIX_MUSIC="${CMAKE_CURRENT_SOURCE_DIR}/assets/music/luminous-flow.wav")
 add_test(NAME audio COMMAND audio_tests)
 set_tests_properties(audio PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
 find_package(OpenGL REQUIRED)
 add_executable(SecondRealityRemix src/main.cpp src/app.cpp src/renderer.cpp)
 target_link_libraries(SecondRealityRemix PRIVATE remix_audio OpenGL::GL)
 if(REMIX_RUNTIME_TESTS)
  add_test(NAME incomplete_control COMMAND ${CMAKE_COMMAND} "-DAPP=$<TARGET_FILE:SecondRealityRemix>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/short_control_test.cmake")
  add_test(NAME lifecycle COMMAND SecondRealityRemix --self-test --duration 15)
  set_tests_properties(incomplete_control lifecycle PROPERTIES TIMEOUT 30)
 endif()
 add_custom_command(TARGET SecondRealityRemix POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:SecondRealityRemix>/assets")
 install(TARGETS SecondRealityRemix RUNTIME DESTINATION .)
 install(DIRECTORY assets DESTINATION .)
 install(DIRECTORY licenses DESTINATION . OPTIONAL)
 install(FILES README.md DESTINATION . OPTIONAL)
 install(FILES LICENSE DESTINATION .)
 install(FILES music/asset-receipt.json music/analysis.json DESTINATION music)
 install(FILES docs/SOURCE_MAP.md docs/VERIFICATION.md DESTINATION docs OPTIONAL)
endif()
